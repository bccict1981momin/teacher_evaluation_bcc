function doGet() { const template = HtmlService.createTemplateFromFile('Index'); template.logoDataUri = getLogoDataUri(); return template.evaluate() .setTitle('Teachers Evaluation'); } function doPost(event) { try { if (!event || !event.postData || !event.postData.contents) { throw new Error('No submission data was received.'); } const data = JSON.parse(event.postData.contents); const message = submitForm(data); return ContentService.createTextOutput(JSON.stringify({ success: true, message: message })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: error.message || 'Submission failed.' })).setMimeType(ContentService.MimeType.JSON); } } function getLogoDataUri() { const imageData = HtmlService.createHtmlOutputFromFile('LogoData') .getContent() .trim(); if (!imageData) { throw new Error('The embedded logo image data is empty.'); } return 'data:image/png;base64,' + imageData; } function submitForm(data) { if (!Array.isArray(data) || data.length === 0) { throw new Error('No teacher evaluations were provided.'); } const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); if (!spreadsheet) { throw new Error('The response spreadsheet could not be opened.'); } const headers = [ 'Submitted At', 'Teacher', 'Subject Knowledge', 'Subject Knowledge Comment', 'Teaching Skill', 'Teaching Skill Comment', 'Sincerity towards the Cadet', 'Sincerity Comment', 'Personality and Behaviour', 'Personality and Behaviour Comment', 'Class', 'House', 'Form', 'Cadet Name', 'Cadet Number' ]; const sheet = getResponseSheet(spreadsheet, 'Responses', headers); const rows = data.map(function(response) { if (!response.teacher) { throw new Error('A teacher name is missing from an evaluation.'); } return [ new Date(), response.teacher || '', response.sk1 || '', response.comment1 || '', response.ts1 || '', response.comment2 || '', response.cm1 || '', response.comment3 || '', response.cs1 || '', response.comment4 || '', response.className || '', response.house || '', response.form || '', response.cadetName || '', response.cadetNumber || '' ]; }); sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, rows[0].length) .setValues(rows); rows.forEach(function(row) { const teacherSheetName = getTeacherSheetName(row[1]); const teacherSheet = getResponseSheet(spreadsheet, teacherSheetName, headers); teacherSheet.getRange(teacherSheet.getLastRow() + 1, 1, 1, row.length) .setValues([row]); }); return 'All ' + rows.length + ' teacher evaluations have been saved to Responses and their individual teacher tabs.'; } function getResponseSheet(spreadsheet, name, headers) { const sheet = spreadsheet.getSheetByName(name) || spreadsheet.insertSheet(name); if (sheet.getLastRow() === 0) { sheet.getRange(1, 1, 1, headers.length).setValues([headers]); } else if (sheet.getRange(1, 1).getValue() !== headers[0]) { sheet.insertRowBefore(1); sheet.getRange(1, 1, 1, headers.length).setValues([headers]); } return sheet; } function getTeacherSheetName(teacherName) { const sheetName = String(teacherName) .replace(/[\[\]:*?\/\\]/g, '') .replace(/^'+|'+$/g, '') .trim() .substring(0, 100); if (!sheetName || sheetName === 'Responses') { throw new Error('The teacher name cannot be used as a sheet name.'); } return sheetName; }
